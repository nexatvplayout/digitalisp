const viewAllowed = (id) => {
  const r = RULES[id];
  if (!r || !state.me) return false;
  if (state.me.mustChange) return id === 'conta';
  if (r.channel && !state.slug) return false;
  // edição Windows: a assinatura é gerida pela central (tela Licença); na nuvem não existe a tela Licença
  if (id === 'licenca' && !state.edge) return false;
  if (id === 'plano' && state.edge) return false;
  // a tela fora do plano do cliente some do menu (as rotas dela também recusam no servidor: server/funcionalidades.js)
  if (state.me.planFeatures && !planViewAllowed(state.me.planFeatures, id)) return false;
  // modo prefeitura (docs/TELAS.md): rede de telas sem comerciais; o menu Comercial some (os dados ficam como estão)
  // (o plano Ilimitado tem tudo ligado, o modo também: nele o Comercial continua)
  if (id === 'comercial' && state.me.planFeatures?.prefeitura && state.me.plan?.code !== 'ilimitado') return false;
  return !r.view || (Array.isArray(r.view) ? r.view.some(can) : can(r.view)); // lista = basta uma das permissões
};

  let st = document.getElementById('planHide');
  const sels = state.me?.planFeatures ? hiddenSelectors(state.me.planFeatures) : [];

  st.textContent = sels.map((s) => `${s} { display: none !important; }`).join('\n');
}

const CONTROLS = 'button, input, select, textarea';
const lockObservers = [];
function applyLocks() {
  lockObservers.splice(0).forEach((o) => o.disconnect());
  $$('.ro').forEach((n) => n.classList.remove('ro'));
  for (const [sel, perm] of LOCKS) {
    if (can(perm)) continue;
    $$(sel).forEach((root) => {
      root.classList.add('ro');
      const shut = () => { [root, ...root.querySelectorAll(CONTROLS)].forEach((c) => { if (c.matches(CONTROLS) && !c.disabled && !c.hasAttribute('data-keep')) { c.disabled = true; c.title = c.title || 'Seu grupo não tem permissão para esta ação.'; } }); };
      shut();
      const o = new MutationObserver(shut);
      o.observe(root, { childList: true, subtree: true, attributes: true, attributeFilter: ['disabled'] });
      lockObservers.push(o);
    });
  }
}

/** Sair de verdade: o servidor revoga ESTE token (POST /api/me/logout; na personificação, o token personificado) e só

async function loadChannels(select) {
  const r = await api('/api/channels');
  state.channels = r.channels.filter((c) => !state.me.channels || state.me.channels.includes(c.slug));
  if (r.fps) state.fps = r.fps;

  state.slug = state.channels.some((c) => c.slug === want) ? want : state.channels[0]?.slug;

    o.value = c.slug; o.textContent = `${c.number} · ${c.name}`; o.selected = c.slug === state.slug;
    return o;

  paintChannelName();
}

/** Telas que mostram o número de espectadores: só com uma delas (ou a janela dos espectadores) o servidor consulta o YouTube. */
const VIEWER_VIEWS = new Set(['monitor', 'playout', 'dashboard']);
async function pollStats() {
  if (!state.slug || !seesLive()) return;
  try {
    // aba escondida (outra aba do navegador, janela minimizada): ninguém está vendo o número, o YouTube não é consultado
    const showing = document.visibilityState === 'visible' && (VIEWER_VIEWS.has(current?.id) || viewersDialogOpen());
    const s = await capi(showing ? '/stats?plataformas=1' : '/stats');
    state.skew = s.serverTime - Date.now();
    if (s.fps) state.fps = s.fps;
    state.stats = s;
    emit('stats', s);
    paintHealthBadge(s.health);
  } catch { /* próxima tentativa */ }
}

/* ------------------------------------------------------------ mesa AO VIVO: câmera conectada */
// câmera do app que chegou com sinal numa posição da mesa AO VIVO, em QUALQUER canal que o usuário vê (GET /api/mesa/cams):
// aviso em qualquer tela, até alguém fechar ou ela sair
const readyCams = new Map(); // canal|posição|usuário -> { pop (null = fechado pelo operador), goneAt }
async function pollCams() {
  if (!state.me || !seesLive()) return;
  if (state.me.planFeatures && !state.me.planFeatures.aovivo) return; // sem a mesa AO VIVO no plano: nada a conferir

  try { paintReadyCams((await api('/api/mesa/cams')).channels || []); } catch { /* próxima tentativa */ }
}
function paintReadyCams(channels) {
  const key = (ch, c) => `${ch.slug}|${c.slot}|${c.user}`;
  const here = new Set(channels.flatMap((ch) => ch.cams.map((c) => key(ch, c))));
  const t = Date.now();
  for (const [k, r] of readyCams) {
    if (here.has(k)) { r.goneAt = 0; continue; }
    // o sinal caiu: o aviso some; uma reconexão rápida não repete o aviso, uma chegada 20 s depois avisa de novo
    r.pop?.remove(); r.pop = null;
    r.goneAt ||= t;
    if (t - r.goneAt > 20000) readyCams.delete(k);
  }
  for (const ch of channels) {
    for (const c of ch.cams) {
      const k = key(ch, c);
      if (readyCams.has(k)) continue;
      const box = $('#readyPops') || document.body.appendChild(el('div', { id: 'readyPops', class: 'ready-pops' }));
      const close = () => { pop.remove(); const r = readyCams.get(k); if (r) r.pop = null; };
      const here_ = ch.slug === state.slug && location.hash.startsWith('#aovivo');
      const pop = el('div', { class: 'ready-pop', role: 'alert' },
        el('span', { class: 'rp-dot', 'aria-hidden': 'true' }),
        el('div', { class: 'rp-t' },
          el('b', { text: `${c.user} se conectou na mesa AO VIVO` }),
          el('small', { text: `${ch.name} · posição ${c.slot + 1}${c.name && c.name !== c.user ? ` (${c.name})` : ''} · a câmera já está na mesa` })),
        el('div', { class: 'rp-acts' },
          // abre a mesa do canal da câmera (troca de canal se preciso, como uma aba)
          here_ ? null : el('button', { class: 'btn small primary', type: 'button', text: 'Abrir a mesa AO VIVO', onclick: () => { close(); emit('open-view', { view: 'aovivo', slug: ch.slug }); } }),
          el('button', { class: 'btn small', type: 'button', text: 'OK', onclick: close })));
      box.append(pop);
      readyCams.set(k, { pop, goneAt: 0 });
    }
  }
}

/* ------------------------------------------------------------ WebSocket (chat, eventos) */
function connectWs() {
  if (ws) return;
  const base = window.NEXA?.apiBase || location.origin;
  const sock = new WebSocket(`${base.replace(/^http/, 'ws')}/ws`);
  ws = sock;
  sock.onopen = () => sock.send(JSON.stringify({ type: 'join', role: 'admin', channel: state.slug, token: state.token })); // sem a sessão, entra como espectador
  sock.onmessage = (ev) => emit('ws', JSON.parse(ev.data));
  sock.onclose = () => { ws = null; if (state.token) setTimeout(connectWs, 3000); };
}

/* ------------------------------------------------------------ roteador */
// endereço "#tela/resto": o resto vai para a tela (ex.: #midia/123 abre os detalhes da mídia 123)
function route(fromHash = false) {
  let [head, ...rest] = location.hash.slice(1).split('/');
  // endereço antigo (abas salvas e links de antes): #plantao virou #aovivo
  if (Object.hasOwn(VIEW_ALIAS, head)) { head = VIEW_ALIAS[head]; history.replaceState(null, '', `#${[head, ...rest].join('/')}`); }
  // aba que mudou de tela: #cg/saidas (Saídas e ajustes) abre Configurações → Gerador de caracteres
  const place = [head, ...rest].join('/');
  if (Object.hasOwn(PLACE_ALIAS, place)) { [head, ...rest] = PLACE_ALIAS[place].split('/'); history.replaceState(null, '', `#${PLACE_ALIAS[place]}`); }
  let id = byId[head] ? head : 'playout';
  let sub = id === head ? decodeURIComponent(rest.join('/')) : '';
  if (!viewAllowed(id)) { id = ORDER.find(viewAllowed) || 'conta'; sub = ''; history.replaceState(null, '', `#${id}`); }
  // só o resto mudou (abrir/fechar detalhes): a tela trata sem recarregar
  if (fromHash === true && current?.id === id && current.sub) { guard(() => current.sub(sub))(); return; }
  if (current && current.id !== id) current.leave?.();
  current = byId[id];
  VIEWS.forEach((v) => { $(`#view-${v.id}`).hidden = v.id !== id; });
  $$('.side a[data-view]').forEach((a) => a.classList.toggle('on', a.dataset.view === id));
  applyLocks();
  guard(async () => { await current.load(sub); applyLocks(); })();
  emit('route', { id, sub }); // tabs.js
}
window.addEventListener('hashchange', () => route(true));

let pollB = null;
async function renderBanner() {
  const items = [];
  const me = state.me;
  if (!me) return;
  if (me.impersonator) items.push({ cls: 'imp', text: `Você está no painel de ${me.tenant?.name || 'um cliente'} como ${me.impersonator} (personificação). Tudo o que fizer fica na auditoria do cliente.` });
  let t = me.tenant;
  if (state.edge) {
    // edição Windows: quem decide é a licença da central (motivo do bloqueio, atraso, sem contato, atualização, mensagens)
    try { for (const b of await edgeBanner()) items.push({ ...b, edge: true }); } catch { /* próxima volta */ }
    t = null;
  } else if (can('billing.view')) { try { t = (await api('/api/billing/plan')).tenant; } catch { /* usa o resumo do /api/me */ } }
  if (t) {
    const soon = t.status === 'trial' && t.trialDaysLeft != null && t.trialDaysLeft <= 5;
    const cls = { past_due: 'warn', suspended: 'bad', canceled: 'bad' }[t.status] || (soon ? 'info' : (t.cancelAtPeriodEnd ? 'info' : null));
    if (cls) items.push({ cls, text: statusLine(t) || 'Verifique o seu plano.', plan: true });
  }
  try {
    const cfg = await (await fetch(`${window.NEXA?.apiBase || ''}/api/config`)).json();
    if (cfg.platform?.announcement) items.push({ cls: cfg.platform.announcement.level === 'warn' ? 'warn' : 'info', text: cfg.platform.announcement.text });
    if (cfg.platform?.maintenance) items.push({ cls: 'warn', text: cfg.platform.maintenance.message || 'Manutenção programada da plataforma.' });
  } catch { /* sem aviso */ }
  // armazenamento a partir de 80% (mudar de plano ou comprar extra); dispensado, volta em 1 dia (80%) ou em poucas horas (95%)
  const storage = state.edge ? null : await storageBanner().catch(() => null);
  $('#tnBanner').replaceChildren(...items.map((b) => el('div', { class: `tn-banner ${b.cls}`, role: 'status' }, el('span', { text: b.text }),
    b.plan && can('billing.view') ? el('a', { href: '#plano', class: 'btn small', text: 'Plano e uso' }) : null,
    b.edge && b.link ? el('a', { href: '#licenca', class: 'btn small', text: 'Licença e suporte' }) : null)), ...(storage ? [storage] : []));
}

      : can('billing.view') ? el('a', { href: '#plano', class: 'btn small', text: 'Ver plano e uso', onclick: () => t.remove() }) : null);
  $('#toasts').append(t);
  setTimeout(() => t.remove(), 9000);
  if (d.code === 'tenant_suspended') renderBanner();
});

}

  state.me = await api('/api/me');

  applyNav();

  await loadMenuNames(); // perfil de nomes dos menus da organização (Padrão NEXA, Broadcast ou um próprio)
  if (state.slug) { connectWs(); await pollStats(); }
  clearInterval(poll);
  poll = setInterval(() => { pollStats(); pollCams(); }, 2000);
  clearInterval(pollO);
  pollOrders(); pollO = setInterval(pollOrders, 15000);
  renderBanner(); clearInterval(pollB); pollB = setInterval(() => renderBanner().catch(() => {}), 60000);
  // volta do pagamento de um pedido da Loja (docs/LOJA.md): a tela da Loja acompanha o pedido até a licença sair
  const lp = new URLSearchParams(location.search);
  if (lp.get('loja_pedido')) {
    state.lojaPedido = { id: Number(lp.get('loja_pedido')) || null, canceled: lp.has('loja_cancelado') };
    history.replaceState(null, '', `${location.pathname}#loja/licencas`);
  }
  // volta do checkout do gateway
  const ck = new URLSearchParams(location.search).get('checkout');
  if (ck) {
    history.replaceState(null, '', `${location.pathname}#plano`);
    toast(ck === 'success' ? 'Pagamento enviado. A confirmação do gateway pode levar alguns instantes.' : 'Pagamento não concluído.', ck !== 'success');
  }
  route();
}
